# minirail Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A personal Railway clone: push to GitHub, get a built and deployed web app, worker or cron job on DigitalOcean Kubernetes, all provisioned by Terraform.

**Architecture:** A thin Go control plane runs in namespace `minirail` and keeps all state in Kubernetes objects (one ConfigMap per app, Secrets for env and database URLs). Builds are Kubernetes Jobs (git clone → Railpack prepare → rootless BuildKit) that push to DigitalOcean Container Registry (DOCR); a 10 s reconcile loop rolls out the newest successful build. Terraform runs in two stages: `infra/cloud` (cluster, registry, DNS, Let's Encrypt cert, managed Postgres/Valkey) and `infra/platform` (Traefik behind the DO load balancer, Secrets, the control plane).

**Tech Stack:** Go 1.26 (client-go v0.37.1, pgx v5.11.0, robfig/cron v3.0.1), `html/template`, Terraform 1.16.4 in Docker (digitalocean 2.102.0, kubernetes 3.2.1, helm 3.3.0, random 3.9.1), Traefik chart 41.6.1, Railpack v0.40.1, `moby/buildkit:v0.33.1-rootless`.

**Spec:** `docs/superpowers/specs/2026-10-15-railway-clone-design.md`

```go
package apps

import (
	"fmt"
	"strings"
)

type Config struct {
	Namespace        string
	Domain           string
	Registry         string
	ControlImage     string
	BuildkitImage    string
	RailpackFrontend string
	GitHubToken      string
	WebhookSecret    string
	AdminPassword    string
	PGAdminURL       string
	ValkeyEnabled    bool
	Addr             string
}

func LoadConfig(getenv func(string) string) (Config, error) {
	valueOr := func(key, fallback string) string {
		if value := getenv(key); value != "" {
			return value
		}
		return fallback
	}
	config := Config{
		Namespace:        valueOr("MINIRAIL_NAMESPACE", "apps"),
		Domain:           getenv("MINIRAIL_DOMAIN"),
		Registry:         getenv("MINIRAIL_REGISTRY"),
		ControlImage:     getenv("MINIRAIL_CONTROL_IMAGE"),
		BuildkitImage:    valueOr("MINIRAIL_BUILDKIT_IMAGE", "moby/buildkit:v0.33.1-rootless"),
		RailpackFrontend: valueOr("MINIRAIL_RAILPACK_FRONTEND", "ghcr.io/railwayapp/railpack-frontend:v0.40.1"),
		GitHubToken:      getenv("GITHUB_TOKEN"),
		WebhookSecret:    getenv("WEBHOOK_SECRET"),
		AdminPassword:    getenv("ADMIN_PASSWORD"),
		PGAdminURL:       getenv("PG_ADMIN_URL"),
		ValkeyEnabled:    getenv("MINIRAIL_VALKEY") == "true",
		Addr:             valueOr("MINIRAIL_ADDR", ":8080"),
	}
	var missing []string
	for _, required := range []struct{ key, value string }{
		{"MINIRAIL_DOMAIN", config.Domain},
		{"MINIRAIL_REGISTRY", config.Registry},
		{"MINIRAIL_CONTROL_IMAGE", config.ControlImage},
		{"GITHUB_TOKEN", config.GitHubToken},
		{"WEBHOOK_SECRET", config.WebhookSecret},
		{"ADMIN_PASSWORD", config.AdminPassword},
	} {
		if required.value == "" {
			missing = append(missing, required.key)
		}
	}
	if len(missing) > 0 {
		return Config{}, fmt.Errorf("missing required env: %s", strings.Join(missing, ", "))
	}
	return config, nil
}
```

```go
	Worker Type = "worker"
	Cron   Type = "cron"
)

const (
	LabelApp              = "paas/app"
	LabelKind             = "paas/kind"
	AnnotationSHA         = "paas/sha"
	AnnotationRestartedAt = "paas/restarted-at"
)

type App struct {
	Name        string
	Repo        string
	Branch      string
	Type        Type
	Port        int
	Schedule    string
	Replicas    int
	Root        string
	Postgres    bool
	Valkey      bool
	DeployedSHA string
}

var (
	namePattern   = regexp.MustCompile(`^[a-z0-9]([-a-z0-9]*[a-z0-9])?$`)
	repoPattern   = regexp.MustCompile(`^https://github\.com/[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$`)
	branchPattern = regexp.MustCompile(`^[A-Za-z0-9._/-]+$`)
	rootPattern   = regexp.MustCompile(`^[A-Za-z0-9._/-]+$`)
	envKeyPattern = regexp.MustCompile(`^[A-Za-z_][A-Za-z0-9_]*$`)
	shaPattern    = regexp.MustCompile(`^[0-9a-f]{40}$`)
	reservedNames = []string{"console", "www", "minirail"}
)

func Labels(name string) map[string]string {
	return map[string]string{LabelApp: name}
}

func ConfigMapName(name string) string { return name + "-app" }
func EnvSecretName(name string) string { return name + "-env" }
func DBSecretName(name string) string  { return name + "-db" }

func (a App) Validate() error {
	switch {
	case len(a.Name) > 30 || !namePattern.MatchString(a.Name):
		return errors.New("name must be 1-30 lowercase letters, digits or dashes, starting and ending with a letter or digit")
	case slices.Contains(reservedNames, a.Name):
		return fmt.Errorf("name %q is reserved", a.Name)
	case !repoPattern.MatchString(a.Repo):
		return errors.New("repo must look like https://github.com/<owner>/<repo>")
	case len(a.Branch) > 100 || !branchPattern.MatchString(a.Branch) || strings.Contains(a.Branch, "..") || strings.HasPrefix(a.Branch, "-"):
		return errors.New("branch may only contain letters, digits, '.', '_', '/' and '-'")
	case !rootPattern.MatchString(a.Root) || path.IsAbs(a.Root) || slices.Contains(strings.Split(a.Root, "/"), ".."):
		return errors.New("root must be a relative path inside the repo")
	}
	switch a.Type {
	case Web:
		if a.Port < 1 || a.Port > 65535 {
			return errors.New("port must be between 1 and 65535")
		}
		return validateReplicas(a.Replicas)
	case Worker:
		return validateReplicas(a.Replicas)
	case Cron:
		return validateSchedule(a.Schedule)
	}
	return errors.New("type must be web, worker or cron")
}

func validateReplicas(replicas int) error {
	if replicas < 1 || replicas > 5 {
		return errors.New("replicas must be between 1 and 5")
	}
	return nil
}

func validateSchedule(schedule string) error {
	if len(strings.Fields(schedule)) != 5 {
		return errors.New("schedule must have exactly 5 fields, e.g. */15 * * * *")
	}
	if _, err := cron.ParseStandard(schedule); err != nil {
		return fmt.Errorf("schedule: %w", err)
	}
	return nil
}

func NormalizeRepo(repo string) string {
	repo = strings.TrimRight(strings.ToLower(strings.TrimSpace(repo)), "/")
	return strings.TrimSuffix(repo, ".git")
}

func ValidEnvKey(key string) bool { return envKeyPattern.MatchString(key) }
func ValidSHA(sha string) bool    { return shaPattern.MatchString(sha) }

func ParseEnv(text string) (map[string]string, error) {
	env := map[string]string{}
	for index, line := range strings.Split(text, "\n") {
		line = strings.TrimRight(line, "\r")
		trimmed := strings.TrimSpace(line)
		if trimmed == "" || strings.HasPrefix(trimmed, "#") {
			continue
		}
		key, value, found := strings.Cut(line, "=")
		key = strings.TrimSpace(key)
		if !found || !ValidEnvKey(key) {
			return nil, fmt.Errorf("line %d: expected KEY=VALUE where KEY is letters, digits and '_'", index+1)
		}
		env[key] = value
	}
	return env, nil
}

func FormatEnv(env map[string]string) string {
	var builder strings.Builder
	for _, key := range slices.Sorted(maps.Keys(env)) {
		builder.WriteString(key + "=" + env[key] + "\n")
	}
	return builder.String()
}

func (a App) ToConfigMap(namespace string) *corev1.ConfigMap {
	return &corev1.ConfigMap{
		ObjectMeta: metav1.ObjectMeta{Name: ConfigMapName(a.Name), Namespace: namespace, Labels: Labels(a.Name)},
		Data: map[string]string{
			"repo":         a.Repo,
			"branch":       a.Branch,
			"type":         string(a.Type),
			"port":         strconv.Itoa(a.Port),
			"schedule":     a.Schedule,
			"replicas":     strconv.Itoa(a.Replicas),
			"root":         a.Root,
			"postgres":     strconv.FormatBool(a.Postgres),
			"valkey":       strconv.FormatBool(a.Valkey),
			"deployed-sha": a.DeployedSHA,
		},
	}
}

func FromConfigMap(configMap *corev1.ConfigMap) (App, error) {
	data := configMap.Data
	port, portErr := strconv.Atoi(data["port"])
	replicas, replicasErr := strconv.Atoi(data["replicas"])
	if err := errors.Join(portErr, replicasErr); err != nil {
		return App{}, fmt.Errorf("app record %s is corrupt: %w", configMap.Name, err)
	}
	return App{
		Name:        configMap.Labels[LabelApp],
		Repo:        data["repo"],
		Branch:      data["branch"],
		Type:        Type(data["type"]),
		Port:        port,
		Schedule:    data["schedule"],
		Replicas:    replicas,
		Root:        data["root"],
		Postgres:    data["postgres"] == "true",
		Valkey:      data["valkey"] == "true",
		DeployedSHA: data["deployed-sha"],
	}, nil
}
```

```go
package apps

import (
	"strconv"

	appsv1 "k8s.io/api/apps/v1"
	batchv1 "k8s.io/api/batch/v1"
	corev1 "k8s.io/api/core/v1"
	networkingv1 "k8s.io/api/networking/v1"
	"k8s.io/apimachinery/pkg/api/resource"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/util/intstr"
	"k8s.io/utils/ptr"
)

const (
	RegistrySecret = "registry"
	ValkeySecret   = "valkey"
)

type Workload struct {
	Deployment *appsv1.Deployment    `json:"deployment,omitempty"`
	Service    *corev1.Service       `json:"service,omitempty"`
	Ingress    *networkingv1.Ingress `json:"ingress,omitempty"`
	CronJob    *batchv1.CronJob      `json:"cronJob,omitempty"`
}

func ImageRef(registry, name, sha string) string {
	return registry + "/" + name + ":" + sha
}

func Host(name, domain string) string { return name + "." + domain }

func BuildWorkload(app App, config Config, image, restartedAt string) Workload {
	meta := metav1.ObjectMeta{Name: app.Name, Namespace: config.Namespace, Labels: Labels(app.Name)}
	template := corev1.PodTemplateSpec{
		ObjectMeta: metav1.ObjectMeta{Labels: Labels(app.Name), Annotations: map[string]string{AnnotationRestartedAt: restartedAt}},
		Spec:       podSpec(app, image),
	}
	if app.Type == Cron {
		template.Spec.RestartPolicy = corev1.RestartPolicyNever
		return Workload{CronJob: &batchv1.CronJob{
			ObjectMeta: meta,
			Spec: batchv1.CronJobSpec{
				Schedule:          app.Schedule,
				ConcurrencyPolicy: batchv1.ForbidConcurrent,
				JobTemplate: batchv1.JobTemplateSpec{
					ObjectMeta: metav1.ObjectMeta{Labels: Labels(app.Name)},
					Spec:       batchv1.JobSpec{BackoffLimit: ptr.To[int32](0), Template: template},
				},
			},
		}}
	}
	workload := Workload{Deployment: &appsv1.Deployment{
		ObjectMeta: meta,
		Spec: appsv1.DeploymentSpec{
			Replicas: ptr.To(int32(app.Replicas)),
			Selector: &metav1.LabelSelector{MatchLabels: Labels(app.Name)},
			Strategy: appsv1.DeploymentStrategy{
				Type: appsv1.RollingUpdateDeploymentStrategyType,
				RollingUpdate: &appsv1.RollingUpdateDeployment{
					MaxUnavailable: ptr.To(intstr.FromInt32(0)),
					MaxSurge:       ptr.To(intstr.FromInt32(1)),
				},
			},
			Template: template,
		},
	}}
	if app.Type == Web {
		workload.Service = &corev1.Service{
			ObjectMeta: meta,
			Spec: corev1.ServiceSpec{
				Selector: Labels(app.Name),
				Ports:    []corev1.ServicePort{{Name: "http", Port: 80, TargetPort: intstr.FromInt32(int32(app.Port))}},
			},
		}
		workload.Ingress = &networkingv1.Ingress{
			ObjectMeta: meta,
			Spec: networkingv1.IngressSpec{
				IngressClassName: ptr.To("traefik"),
				Rules: []networkingv1.IngressRule{{
					Host: Host(app.Name, config.Domain),
					IngressRuleValue: networkingv1.IngressRuleValue{HTTP: &networkingv1.HTTPIngressRuleValue{
						Paths: []networkingv1.HTTPIngressPath{{
							Path:     "/",
							PathType: ptr.To(networkingv1.PathTypePrefix),
							Backend: networkingv1.IngressBackend{Service: &networkingv1.IngressServiceBackend{
								Name: app.Name,
								Port: networkingv1.ServiceBackendPort{Number: 80},
							}},
						}},
					}},
				}},
			},
		}
	}
	return workload
}

func podSpec(app App, image string) corev1.PodSpec {
	container := corev1.Container{
		Name:  "app",
		Image: image,
		EnvFrom: []corev1.EnvFromSource{
			{SecretRef: &corev1.SecretEnvSource{LocalObjectReference: corev1.LocalObjectReference{Name: EnvSecretName(app.Name)}, Optional: ptr.To(true)}},
			{SecretRef: &corev1.SecretEnvSource{LocalObjectReference: corev1.LocalObjectReference{Name: DBSecretName(app.Name)}, Optional: ptr.To(true)}},
		},
		Resources: corev1.ResourceRequirements{
			Requests: corev1.ResourceList{corev1.ResourceCPU: resource.MustParse("100m"), corev1.ResourceMemory: resource.MustParse("128Mi")},
			Limits:   corev1.ResourceList{corev1.ResourceMemory: resource.MustParse("512Mi")},
		},
	}
	if app.Valkey {
		container.EnvFrom = append(container.EnvFrom, corev1.EnvFromSource{
			SecretRef: &corev1.SecretEnvSource{LocalObjectReference: corev1.LocalObjectReference{Name: ValkeySecret}},
		})
	}
	if app.Type == Web {
		container.Env = []corev1.EnvVar{{Name: "PORT", Value: strconv.Itoa(app.Port)}}
		container.Ports = []corev1.ContainerPort{{Name: "http", ContainerPort: int32(app.Port)}}
		container.ReadinessProbe = &corev1.Probe{
			ProbeHandler:  corev1.ProbeHandler{TCPSocket: &corev1.TCPSocketAction{Port: intstr.FromInt32(int32(app.Port))}},
			PeriodSeconds: 5,
		}
	}
	return corev1.PodSpec{
		Containers:       []corev1.Container{container},
		ImagePullSecrets: []corev1.LocalObjectReference{{Name: RegistrySecret}},
	}
}
```

```go
package build

import (
	"context"
	"errors"
	"fmt"
	"os"
	"os/exec"
	"strings"

	"minirail/internal/apps"
)

const credentialHelper = `credential.helper=!f() { echo username=x-access-token; echo "password=$GITHUB_TOKEN"; }; f`

func LsRemote(token string) func(ctx context.Context, repo, branch string) (string, error) {
	return func(ctx context.Context, repo, branch string) (string, error) {
		command := exec.CommandContext(ctx, "git", "-c", credentialHelper, "ls-remote", repo, "refs/heads/"+branch)
		command.Env = append(os.Environ(), "GITHUB_TOKEN="+token, "GIT_TERMINAL_PROMPT=0", "HOME=/tmp")
		output, err := command.Output()
		if exitErr := (*exec.ExitError)(nil); errors.As(err, &exitErr) {
			return "", fmt.Errorf("git ls-remote %s: %s", repo, strings.TrimSpace(string(exitErr.Stderr)))
		}
		if err != nil {
			return "", fmt.Errorf("git ls-remote %s: %w", repo, err)
		}
		return parseLsRemote(string(output), branch)
	}
}

func parseLsRemote(output, branch string) (string, error) {
	for line := range strings.Lines(output) {
		sha, ref, found := strings.Cut(strings.TrimSpace(line), "\t")
		if found && ref == "refs/heads/"+branch && apps.ValidSHA(sha) {
			return sha, nil
		}
	}
	return "", fmt.Errorf("branch %q not found in repo", branch)
}
```

```go
	"context"
	"errors"
```

```go
	"strings"

	apierrors "k8s.io/apimachinery/pkg/api/errors"

	"k8s.io/client-go/kubernetes"
)

var (
	ErrExists   = errors.New("app already exists")
	ErrNotFound = errors.New("app not found")
)

type Store struct {
	Client kubernetes.Interface
	Config Config
}

func (s *Store) Create(ctx context.Context, app App) error {
	_, err := s.Client.CoreV1().ConfigMaps(s.Config.Namespace).Create(ctx, app.ToConfigMap(s.Config.Namespace), metav1.CreateOptions{})
	if apierrors.IsAlreadyExists(err) {
		return fmt.Errorf("%w: %s", ErrExists, app.Name)
	}
	return err
}

func (s *Store) Get(ctx context.Context, name string) (App, error) {
	configMap, err := s.Client.CoreV1().ConfigMaps(s.Config.Namespace).Get(ctx, ConfigMapName(name), metav1.GetOptions{})
	if apierrors.IsNotFound(err) {
		return App{}, fmt.Errorf("%w: %s", ErrNotFound, name)
	}
	if err != nil {
		return App{}, err
	}
	return FromConfigMap(configMap)
}

func (s *Store) List(ctx context.Context) ([]App, error) {
	configMaps, err := s.Client.CoreV1().ConfigMaps(s.Config.Namespace).List(ctx, metav1.ListOptions{LabelSelector: LabelApp})
	if err != nil {
		return nil, err
	}
	var all []App
	for _, configMap := range configMaps.Items {
		app, err := FromConfigMap(&configMap)
		if err != nil {
			return nil, err
		}
		all = append(all, app)
	}
	slices.SortFunc(all, func(left, right App) int { return strings.Compare(left.Name, right.Name) })
	return all, nil
}

func (s *Store) Update(ctx context.Context, app App) error {
	configMaps := s.Client.CoreV1().ConfigMaps(s.Config.Namespace)
	existing, err := configMaps.Get(ctx, ConfigMapName(app.Name), metav1.GetOptions{})
	if err != nil {
		return err
	}
	existing.Data = app.ToConfigMap(s.Config.Namespace).Data
	_, err = configMaps.Update(ctx, existing, metav1.UpdateOptions{})
	return err
}

func (s *Store) Env(ctx context.Context, name string) (map[string]string, error) {
	secret, err := s.Client.CoreV1().Secrets(s.Config.Namespace).Get(ctx, EnvSecretName(name), metav1.GetOptions{})
	if apierrors.IsNotFound(err) {
		return map[string]string{}, nil
	}
	if err != nil {
		return nil, err
	}
	env := map[string]string{}
	for key, value := range secret.Data {
		env[key] = string(value)
	}
	return env, nil
}

func (s *Store) SetEnv(ctx context.Context, name string, env map[string]string) error {
	return s.upsertSecret(ctx, name, EnvSecretName(name), env)
}

func (s *Store) SetDatabaseURL(ctx context.Context, name, databaseURL string) error {
	return s.upsertSecret(ctx, name, DBSecretName(name), map[string]string{"DATABASE_URL": databaseURL})
}

func (s *Store) upsertSecret(ctx context.Context, appName, secretName string, values map[string]string) error {
	data := map[string][]byte{}
	for key, value := range values {
		data[key] = []byte(value)
	}
	secrets := s.Client.CoreV1().Secrets(s.Config.Namespace)
	existing, err := secrets.Get(ctx, secretName, metav1.GetOptions{})
	if apierrors.IsNotFound(err) {
		_, err = secrets.Create(ctx, &corev1.Secret{
			ObjectMeta: metav1.ObjectMeta{Name: secretName, Namespace: s.Config.Namespace, Labels: Labels(appName)},
			Data:       data,
		}, metav1.CreateOptions{})
		return err
	}
	if err != nil {
		return err
	}
	existing.Data = data
	_, err = secrets.Update(ctx, existing, metav1.UpdateOptions{})
	return err
}
```

```go
package apps

import (
	"context"
	"errors"
	"fmt"
	"time"

	batchv1 "k8s.io/api/batch/v1"
	apierrors "k8s.io/apimachinery/pkg/api/errors"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
)

var ErrNotDeployed = errors.New("app has not been deployed yet")

func (s *Store) Apply(ctx context.Context, app App, image string) error {
	workload := BuildWorkload(app, s.Config, image, time.Now().UTC().Format(time.RFC3339))
	namespace := s.Config.Namespace
	if desired := workload.Deployment; desired != nil {
		deployments := s.Client.AppsV1().Deployments(namespace)
		existing, err := deployments.Get(ctx, desired.Name, metav1.GetOptions{})
		switch {
		case apierrors.IsNotFound(err):
			_, err = deployments.Create(ctx, desired, metav1.CreateOptions{})
		case err == nil:
			existing.Spec = desired.Spec
			_, err = deployments.Update(ctx, existing, metav1.UpdateOptions{})
		}
		if err != nil {
			return fmt.Errorf("deployment %s: %w", desired.Name, err)
		}
	}
	if desired := workload.CronJob; desired != nil {
		cronJobs := s.Client.BatchV1().CronJobs(namespace)
		existing, err := cronJobs.Get(ctx, desired.Name, metav1.GetOptions{})
		switch {
		case apierrors.IsNotFound(err):
			_, err = cronJobs.Create(ctx, desired, metav1.CreateOptions{})
		case err == nil:
			existing.Spec = desired.Spec
			_, err = cronJobs.Update(ctx, existing, metav1.UpdateOptions{})
		}
		if err != nil {
			return fmt.Errorf("cronjob %s: %w", desired.Name, err)
		}
	}
	if desired := workload.Service; desired != nil {
		_, err := s.Client.CoreV1().Services(namespace).Create(ctx, desired, metav1.CreateOptions{})
		if err != nil && !apierrors.IsAlreadyExists(err) {
			return fmt.Errorf("service %s: %w", desired.Name, err)
		}
	}
	if desired := workload.Ingress; desired != nil {
		_, err := s.Client.NetworkingV1().Ingresses(namespace).Create(ctx, desired, metav1.CreateOptions{})
		if err != nil && !apierrors.IsAlreadyExists(err) {
			return fmt.Errorf("ingress %s: %w", desired.Name, err)
		}
	}
	return nil
}

func (s *Store) Restart(ctx context.Context, app App) error {
	if app.DeployedSHA == "" {
		return nil
	}
	return s.Apply(ctx, app, ImageRef(s.Config.Registry, app.Name, app.DeployedSHA))
}

func (s *Store) RunNow(ctx context.Context, app App) (string, error) {
	cronJob, err := s.Client.BatchV1().CronJobs(s.Config.Namespace).Get(ctx, app.Name, metav1.GetOptions{})
	if apierrors.IsNotFound(err) {
		return "", ErrNotDeployed
	}
	if err != nil {
		return "", err
	}
	job := &batchv1.Job{
		ObjectMeta: metav1.ObjectMeta{
			Name:      fmt.Sprintf("%s-run-%d", app.Name, time.Now().Unix()),
			Namespace: s.Config.Namespace,
			Labels:    Labels(app.Name),
		},
		Spec: cronJob.Spec.JobTemplate.Spec,
	}
	created, err := s.Client.BatchV1().Jobs(s.Config.Namespace).Create(ctx, job, metav1.CreateOptions{})
	if err != nil {
		return "", err
	}
	return created.Name, nil
}
```

```go
package apps

import (
	"context"
	"errors"

	apierrors "k8s.io/apimachinery/pkg/api/errors"
	"k8s.io/apimachinery/pkg/api/meta"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/runtime"
	"k8s.io/utils/ptr"
)

func (s *Store) Delete(ctx context.Context, name string) error {
	namespace := s.Config.Namespace
	selector := metav1.ListOptions{LabelSelector: LabelApp + "=" + name}
	background := metav1.DeleteOptions{PropagationPolicy: ptr.To(metav1.DeletePropagationBackground)}
	deployments, services, ingresses := s.Client.AppsV1().Deployments(namespace), s.Client.CoreV1().Services(namespace), s.Client.NetworkingV1().Ingresses(namespace)
	cronJobs, jobs := s.Client.BatchV1().CronJobs(namespace), s.Client.BatchV1().Jobs(namespace)
	secrets, configMaps := s.Client.CoreV1().Secrets(namespace), s.Client.CoreV1().ConfigMaps(namespace)
	kinds := []struct {
		list   func() (runtime.Object, error)
		remove func(context.Context, string, metav1.DeleteOptions) error
	}{
		{func() (runtime.Object, error) { return deployments.List(ctx, selector) }, deployments.Delete},
		{func() (runtime.Object, error) { return services.List(ctx, selector) }, services.Delete},
		{func() (runtime.Object, error) { return ingresses.List(ctx, selector) }, ingresses.Delete},
		{func() (runtime.Object, error) { return cronJobs.List(ctx, selector) }, cronJobs.Delete},
		{func() (runtime.Object, error) { return jobs.List(ctx, selector) }, jobs.Delete},
		{func() (runtime.Object, error) { return secrets.List(ctx, selector) }, secrets.Delete},
		// The app record goes last so a partial failure leaves the app listed and the delete retryable.
		{func() (runtime.Object, error) { return configMaps.List(ctx, selector) }, configMaps.Delete},
	}
	var errs []error
	for _, kind := range kinds {
		list, err := kind.list()
		if err != nil {
			errs = append(errs, err)
			continue
		}
		items, err := meta.ExtractList(list)
		if err != nil {
			errs = append(errs, err)
			continue
		}
		for _, item := range items {
			object, err := meta.Accessor(item)
			if err == nil {
				err = kind.remove(ctx, object.GetName(), background)
			}
			if err != nil && !apierrors.IsNotFound(err) {
				errs = append(errs, err)
			}
		}
	}
	return errors.Join(errs...)
}
```

```go
package build

import (
	"context"
	"slices"
	"strings"

	"minirail/internal/apps"

	batchv1 "k8s.io/api/batch/v1"
	corev1 "k8s.io/api/core/v1"
	apierrors "k8s.io/apimachinery/pkg/api/errors"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/client-go/kubernetes"
	"k8s.io/utils/ptr"
)

const (
	StatusRunning   = "running"
	StatusSucceeded = "succeeded"
	StatusFailed    = "failed"
)

type Builder struct {
	Client      kubernetes.Interface
	Config      apps.Config
	Store       *apps.Store
	ResolveHead func(ctx context.Context, repo, branch string) (string, error)
}

func Status(job batchv1.Job) string {
	for _, condition := range job.Status.Conditions {
		if condition.Status != corev1.ConditionTrue {
			continue
		}
		switch condition.Type {
		case batchv1.JobComplete:
			return StatusSucceeded
		case batchv1.JobFailed:
			return StatusFailed
		}
	}
	return StatusRunning
}

func (b *Builder) Trigger(ctx context.Context, app apps.App, sha string) (bool, error) {
	if sha == app.DeployedSHA {
		return false, nil
	}
	jobs := b.Client.BatchV1().Jobs(b.Config.Namespace)
	name := JobName(app.Name, sha)
	existing, err := jobs.Get(ctx, name, metav1.GetOptions{})
	switch {
	case err == nil && Status(*existing) == StatusRunning:
		return false, nil
	case err == nil:
		// A finished Job is rebuilt rather than reused so it becomes the newest build and Reconcile deploys it.
		background := metav1.DeleteOptions{PropagationPolicy: ptr.To(metav1.DeletePropagationBackground)}
		if err := jobs.Delete(ctx, name, background); err != nil && !apierrors.IsNotFound(err) {
			return false, err
		}
	case !apierrors.IsNotFound(err):
		return false, err
	}
	if _, err := jobs.Create(ctx, NewJob(b.Config, app, sha), metav1.CreateOptions{}); err != nil {
		return false, err
	}
	return true, nil
}

func (b *Builder) Redeploy(ctx context.Context, app apps.App) (bool, error) {
	sha, err := b.ResolveHead(ctx, app.Repo, app.Branch)
	if err != nil {
		return false, err
	}
	return b.Trigger(ctx, app, sha)
}

func (b *Builder) Builds(ctx context.Context, appName string) ([]batchv1.Job, error) {
	selector := apps.LabelApp + "=" + appName + "," + apps.LabelKind + "=" + KindBuild
	list, err := b.Client.BatchV1().Jobs(b.Config.Namespace).List(ctx, metav1.ListOptions{LabelSelector: selector})
	if err != nil {
		return nil, err
	}
	slices.SortFunc(list.Items, newestFirst)
	return list.Items, nil
}

func newestFirst(left, right batchv1.Job) int {
	if order := right.CreationTimestamp.Compare(left.CreationTimestamp.Time); order != 0 {
		return order
	}
	return strings.Compare(right.Name, left.Name)
}
```

```go
package build

import (
	"context"
	"errors"
	"log"
	"time"

	"minirail/internal/apps"

	batchv1 "k8s.io/api/batch/v1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
)

func (b *Builder) Reconcile(ctx context.Context) error {
	list, err := b.Client.BatchV1().Jobs(b.Config.Namespace).List(ctx, metav1.ListOptions{LabelSelector: apps.LabelKind + "=" + KindBuild})
	if err != nil {
		return err
	}
	newestSucceeded := map[string]batchv1.Job{}
	for _, job := range list.Items {
		if Status(job) != StatusSucceeded {
			continue
		}
		appName := job.Labels[apps.LabelApp]
		if current, seen := newestSucceeded[appName]; !seen || newestFirst(job, current) < 0 {
			newestSucceeded[appName] = job
		}
	}
	var errs []error
	for appName, job := range newestSucceeded {
		sha := job.Annotations[apps.AnnotationSHA]
		app, err := b.Store.Get(ctx, appName)
		if errors.Is(err, apps.ErrNotFound) || !apps.ValidSHA(sha) {
			continue
		}
		if err != nil {
			errs = append(errs, err)
			continue
		}
		if sha == app.DeployedSHA {
			continue
		}
		if err := b.Store.Apply(ctx, app, apps.ImageRef(b.Config.Registry, app.Name, sha)); err != nil {
			errs = append(errs, err)
			continue
		}
		app.DeployedSHA = sha
		if err := b.Store.Update(ctx, app); err != nil {
			errs = append(errs, err)
		}
	}
	return errors.Join(errs...)
}

func (b *Builder) Run(ctx context.Context, interval time.Duration) {
	ticker := time.NewTicker(interval)
	defer ticker.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
			if err := b.Reconcile(ctx); err != nil {
				log.Printf("reconcile: %v", err)
			}
		}
	}
}
```

```go
package pg

import (
	"context"
	"crypto/rand"
	"fmt"
	"net/url"
	"strings"

	"github.com/jackc/pgx/v5"
)

type Provisioner struct {
	AdminURL string
}

func RoleName(appName string) string { return "app_" + strings.ReplaceAll(appName, "-", "_") }

func (p Provisioner) Provision(ctx context.Context, appName string) (string, error) {
	role, password := RoleName(appName), rand.Text()
	conn, err := pgx.Connect(ctx, p.AdminURL)
	if err != nil {
		return "", fmt.Errorf("connect to postgres: %w", err)
	}
	defer conn.Close(ctx)
	var roleExists, databaseExists bool
	err = conn.QueryRow(ctx,
		"SELECT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = $1), EXISTS (SELECT 1 FROM pg_database WHERE datname = $1)",
		role).Scan(&roleExists, &databaseExists)
	if err != nil {
		return "", err
	}
	for _, statement := range statements(role, password, roleExists, databaseExists) {
		if _, err := conn.Exec(ctx, statement); err != nil {
			return "", fmt.Errorf("%s: %w", strings.Fields(statement)[0], err)
		}
	}
	return appURL(p.AdminURL, role, password)
}

func statements(role, password string, roleExists, databaseExists bool) []string {
	quoted := pgx.Identifier{role}.Sanitize()
	list := []string{fmt.Sprintf("CREATE ROLE %s LOGIN PASSWORD '%s'", quoted, password)}
	if roleExists {
		list = []string{fmt.Sprintf("ALTER ROLE %s WITH LOGIN PASSWORD '%s'", quoted, password)}
	}
	// The managed admin is not a superuser, so it must be a member of the role to make it a database owner.
	list = append(list, fmt.Sprintf("GRANT %s TO CURRENT_USER", quoted))
	if !databaseExists {
		list = append(list, fmt.Sprintf("CREATE DATABASE %s OWNER %s", quoted, quoted))
	}
	return append(list, fmt.Sprintf("REVOKE CONNECT ON DATABASE %s FROM PUBLIC", quoted))
}

func appURL(adminURL, role, password string) (string, error) {
	parsed, err := url.Parse(adminURL)
	if err != nil {
		return "", fmt.Errorf("admin url: %w", err)
	}
	parsed.User = url.UserPassword(role, password)
	parsed.Path = "/" + role
	return parsed.String(), nil
}
```

```go
package web

import (
	"bytes"
	"context"
	"crypto/sha256"
	"crypto/subtle"
	"embed"
	"errors"
	"html/template"
	"log"
	"net/http"

	"minirail/internal/apps"
	"minirail/internal/build"
)

const maxBodyBytes = 1 << 20

var errBadRequest = errors.New("bad request")

//go:embed templates
var templateFiles embed.FS

var templateFuncs = template.FuncMap{
	"short": func(sha string) string {
		if len(sha) > 7 {
			return sha[:7]
		}
		return sha
	},
	"buildStatus": build.Status,
}

func page(name string) *template.Template {
	return template.Must(template.New(name).Funcs(templateFuncs).ParseFS(templateFiles, "templates/layout.html", "templates/"+name+".html"))
}

type Server struct {
	Config    apps.Config
	Store     *apps.Store
	Builder   *build.Builder
	Provision func(ctx context.Context, appName string) (string, error)
}

func (s *Server) Handler() http.Handler {
	dashboard := http.NewServeMux()

	root := http.NewServeMux()
	root.HandleFunc("GET /healthz", func(w http.ResponseWriter, _ *http.Request) { w.Write([]byte("ok")) })
	root.Handle("/", s.requireAdmin(http.NewCrossOriginProtection().Handler(dashboard)))
	return http.MaxBytesHandler(root, maxBodyBytes)
}

func (s *Server) requireAdmin(next http.Handler) http.Handler {
	want := sha256.Sum256([]byte("admin:" + s.Config.AdminPassword))
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		user, password, _ := r.BasicAuth()
		got := sha256.Sum256([]byte(user + ":" + password))
		if subtle.ConstantTimeCompare(got[:], want[:]) != 1 {
			w.Header().Set("WWW-Authenticate", `Basic realm="minirail", charset="UTF-8"`)
			http.Error(w, "unauthorized", http.StatusUnauthorized)
			return
		}
		next.ServeHTTP(w, r)
	})
}

func render(w http.ResponseWriter, status int, tmpl *template.Template, data any) {
	var body bytes.Buffer
	if err := tmpl.ExecuteTemplate(&body, "layout", data); err != nil {
		log.Printf("render %s: %v", tmpl.Name(), err)
		http.Error(w, "render failed", http.StatusInternalServerError)
		return
	}
	w.Header().Set("Content-Type", "text/html; charset=utf-8")
	w.WriteHeader(status)
	body.WriteTo(w)
}

func fail(w http.ResponseWriter, err error) {
	switch {
	case errors.Is(err, errBadRequest):
		http.Error(w, err.Error(), http.StatusBadRequest)
	case errors.Is(err, apps.ErrNotFound):
		http.Error(w, err.Error(), http.StatusNotFound)
	case errors.Is(err, apps.ErrExists), errors.Is(err, apps.ErrNotDeployed):
		http.Error(w, err.Error(), http.StatusConflict)
	default:
		log.Printf("cluster error: %v", err)
		http.Error(w, "cluster request failed: "+err.Error(), http.StatusBadGateway)
	}
}

func (s *Server) appURL(app apps.App) string {
	if app.Type != apps.Web {
		return ""
	}
	return "https://" + apps.Host(app.Name, s.Config.Domain)
}
```

```go
package web

import (
	"errors"
	"log"
	"net/http"
	"strconv"
	"strings"

	"minirail/internal/apps"
	"minirail/internal/build"
)

var (
	indexPage  = page("index")
	newAppPage = page("new")
)

type appRow struct {
	App       apps.App
	URL       string
	LastBuild string
}

type newAppView struct {
	App   apps.App
	Error string
}

func (s *Server) index(w http.ResponseWriter, r *http.Request) {
	list, err := s.Store.List(r.Context())
	if err != nil {
		fail(w, err)
		return
	}
	rows := make([]appRow, 0, len(list))
	for _, app := range list {
		builds, err := s.Builder.Builds(r.Context(), app.Name)
		if err != nil {
			fail(w, err)
			return
		}
		row := appRow{App: app, URL: s.appURL(app), LastBuild: "none"}
		if len(builds) > 0 {
			row.LastBuild = build.Status(builds[0])
		}
		rows = append(rows, row)
	}
	render(w, http.StatusOK, indexPage, rows)
}

func (s *Server) newForm(w http.ResponseWriter, r *http.Request) {
	defaults := apps.App{Branch: "main", Type: apps.Web, Port: 8080, Replicas: 1, Root: "."}
	render(w, http.StatusOK, newAppPage, newAppView{App: defaults})
}

func (s *Server) create(w http.ResponseWriter, r *http.Request) {
	app := appFromForm(r)
	if err := app.Validate(); err != nil {
		render(w, http.StatusBadRequest, newAppPage, newAppView{App: app, Error: err.Error()})
		return
	}
	if err := s.Store.Create(r.Context(), app); errors.Is(err, apps.ErrExists) {
		render(w, http.StatusConflict, newAppPage, newAppView{App: app, Error: err.Error()})
		return
	} else if err != nil {
		fail(w, err)
		return
	}
	if _, err := s.Builder.Redeploy(r.Context(), app); err != nil {
		log.Printf("first build of %s: %v", app.Name, err)
		http.Error(w, "App created, but the first build could not start: "+err.Error()+". Press Deploy on the app page to retry.", http.StatusBadGateway)
		return
	}
	http.Redirect(w, r, "/apps/"+app.Name, http.StatusSeeOther)
}

func appFromForm(r *http.Request) apps.App {
	number := func(key string) int {
		value, _ := strconv.Atoi(r.PostFormValue(key))
		return value
	}
	return apps.App{
		Name:     strings.TrimSpace(r.PostFormValue("name")),
		Repo:     apps.NormalizeRepo(r.PostFormValue("repo")),
		Branch:   strings.TrimSpace(r.PostFormValue("branch")),
		Type:     apps.Type(r.PostFormValue("type")),
		Port:     number("port"),
		Schedule: strings.TrimSpace(r.PostFormValue("schedule")),
		Replicas: number("replicas"),
		Root:     strings.TrimSpace(r.PostFormValue("root")),
	}
}
```

```go
package web

import (
	"net/http"

	"minirail/internal/apps"
	"minirail/internal/build"

	batchv1 "k8s.io/api/batch/v1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
)

var detailPage = page("detail")

type podRow struct {
	Name     string
	Phase    string
	Restarts int32
}

type detailView struct {
	App         apps.App
	URL         string
	Env         string
	Builds      []batchv1.Job
	Pods        []podRow
	WebhookURL  string
	CanPostgres bool
	CanValkey   bool
}

func (s *Server) detail(w http.ResponseWriter, r *http.Request) {
	ctx := r.Context()
	app, err := s.Store.Get(ctx, r.PathValue("name"))
	if err != nil {
		fail(w, err)
		return
	}
	env, err := s.Store.Env(ctx, app.Name)
	if err != nil {
		fail(w, err)
		return
	}
	builds, err := s.Builder.Builds(ctx, app.Name)
	if err != nil {
		fail(w, err)
		return
	}
	pods, err := s.Store.Client.CoreV1().Pods(s.Config.Namespace).List(ctx, metav1.ListOptions{LabelSelector: runtimeSelector(app.Name)})
	if err != nil {
		fail(w, err)
		return
	}
	podRows := make([]podRow, 0, len(pods.Items))
	for _, pod := range pods.Items {
		row := podRow{Name: pod.Name, Phase: string(pod.Status.Phase)}
		for _, container := range pod.Status.ContainerStatuses {
			row.Restarts += container.RestartCount
		}
		podRows = append(podRows, row)
	}
	render(w, http.StatusOK, detailPage, detailView{
		App:         app,
		URL:         s.appURL(app),
		Env:         apps.FormatEnv(env),
		Builds:      builds,
		Pods:        podRows,
		WebhookURL:  "https://" + apps.Host("console", s.Config.Domain) + "/webhook/github",
		CanPostgres: s.Provision != nil,
		CanValkey:   s.Config.ValkeyEnabled,
	})
}

func runtimeSelector(appName string) string {
	return apps.LabelApp + "=" + appName + "," + apps.LabelKind + "!=" + build.KindBuild
}
```

```go
package web

import (
	"context"
	"fmt"
	"net/http"

	"minirail/internal/apps"
)

func (s *Server) appAction(handle func(r *http.Request, app apps.App) error) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		app, err := s.Store.Get(r.Context(), r.PathValue("name"))
		if err == nil {
			err = handle(r, app)
		}
		if err != nil {
			fail(w, err)
			return
		}
		http.Redirect(w, r, "/apps/"+app.Name, http.StatusSeeOther)
	}
}

func (s *Server) saveEnv(r *http.Request, app apps.App) error {
	env, err := apps.ParseEnv(r.PostFormValue("env"))
	if err != nil {
		return fmt.Errorf("%w: %w", errBadRequest, err)
	}
	if err := s.Store.SetEnv(r.Context(), app.Name, env); err != nil {
		return err
	}
	return s.Store.Restart(r.Context(), app)
}

func (s *Server) deploy(r *http.Request, app apps.App) error {
	_, err := s.Builder.Redeploy(r.Context(), app)
	return err
}

func (s *Server) run(r *http.Request, app apps.App) error {
	if app.Type != apps.Cron {
		return fmt.Errorf("%w: only cron apps can run on demand", errBadRequest)
	}
	_, err := s.Store.RunNow(r.Context(), app)
	return err
}

func (s *Server) attachPostgres(r *http.Request, app apps.App) error {
	if s.Provision == nil {
		return fmt.Errorf("%w: postgres is not configured", errBadRequest)
	}
	databaseURL, err := s.Provision(r.Context(), app.Name)
	if err != nil {
		return fmt.Errorf("provision postgres for %s: %w", app.Name, err)
	}
	if err := s.Store.SetDatabaseURL(r.Context(), app.Name, databaseURL); err != nil {
		return err
	}
	app.Postgres = true
	return s.saveAndRestart(r.Context(), app)
}

func (s *Server) attachValkey(r *http.Request, app apps.App) error {
	if !s.Config.ValkeyEnabled {
		return fmt.Errorf("%w: valkey is not enabled", errBadRequest)
	}
	app.Valkey = true
	return s.saveAndRestart(r.Context(), app)
}

func (s *Server) saveAndRestart(ctx context.Context, app apps.App) error {
	if err := s.Store.Update(ctx, app); err != nil {
		return err
	}
	return s.Store.Restart(ctx, app)
}

func (s *Server) remove(w http.ResponseWriter, r *http.Request) {
	if err := s.Store.Delete(r.Context(), r.PathValue("name")); err != nil {
		fail(w, err)
		return
	}
	http.Redirect(w, r, "/", http.StatusSeeOther)
}
```

```go
package web

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"log"
	"net/http"
	"strings"

	"minirail/internal/apps"
)

type pushEvent struct {
	Ref        string `json:"ref"`
	After      string `json:"after"`
	Deleted    bool   `json:"deleted"`
	Repository struct {
		HTMLURL string `json:"html_url"`
	} `json:"repository"`
}

func VerifySignature(secret string, body []byte, header string) bool {
	signature, found := strings.CutPrefix(header, "sha256=")
	if !found {
		return false
	}
	given, err := hex.DecodeString(signature)
	if err != nil {
		return false
	}
	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write(body)
	return hmac.Equal(given, mac.Sum(nil))
}

func (s *Server) webhook(w http.ResponseWriter, r *http.Request) {
	body, err := io.ReadAll(r.Body)
	if _, tooLarge := errors.AsType[*http.MaxBytesError](err); tooLarge {
		http.Error(w, "payload too large: press Deploy on the app page", http.StatusRequestEntityTooLarge)
		return
	}
	if err != nil {
		http.Error(w, "read body", http.StatusBadRequest)
		return
	}
	if !VerifySignature(s.Config.WebhookSecret, body, r.Header.Get("X-Hub-Signature-256")) {
		log.Printf("webhook: bad signature from %s", r.RemoteAddr)
		http.Error(w, "bad signature", http.StatusUnauthorized)
		return
	}
	switch r.Header.Get("X-GitHub-Event") {
	case "ping":
		w.WriteHeader(http.StatusOK)
		return
	case "push":
	default:
		w.WriteHeader(http.StatusNoContent)
		return
	}
	var event pushEvent
	if err := json.Unmarshal(body, &event); err != nil {
		http.Error(w, "invalid push payload", http.StatusBadRequest)
		return
	}
	branch, isBranch := strings.CutPrefix(event.Ref, "refs/heads/")
	if event.Deleted || !isBranch {
		w.WriteHeader(http.StatusNoContent)
		return
	}
	if !apps.ValidSHA(event.After) {
		http.Error(w, "invalid after sha", http.StatusBadRequest)
		return
	}
	list, err := s.Store.List(r.Context())
	if err != nil {
		fail(w, err)
		return
	}
	repo := apps.NormalizeRepo(event.Repository.HTMLURL)
	var matched int
	var errs []error
	for _, app := range list {
		if app.Repo != repo || app.Branch != branch {
			continue
		}
		matched++
		if _, err := s.Builder.Trigger(r.Context(), app, event.After); err != nil {
			errs = append(errs, fmt.Errorf("%s: %w", app.Name, err))
		}
	}
	if err := errors.Join(errs...); err != nil {
		fail(w, err)
		return
	}
	if matched == 0 {
		w.WriteHeader(http.StatusNoContent)
		return
	}
	w.WriteHeader(http.StatusAccepted)
}
```

```go
	"context"
	"errors"
	"fmt"

	batchv1 "k8s.io/api/batch/v1"

)

var errNoLogs = errors.New("no pod to read logs from yet")

var buildContainers = []string{"clone", "prepare", "build"}

func (s *Server) logs(w http.ResponseWriter, r *http.Request) {
	ctx := r.Context()
	app, err := s.Store.Get(ctx, r.PathValue("name"))
	if err != nil {
		fail(w, err)
		return
	}
	podName, containers, err := s.logTarget(ctx, app, r.URL.Query().Get("source"))
	if err != nil {
		fail(w, err)
		return
	}
	w.Header().Set("Content-Type", "text/event-stream")
	w.Header().Set("Cache-Control", "no-cache")
	flusher := http.NewResponseController(w)
	pods := s.Store.Client.CoreV1().Pods(s.Config.Namespace)
	for _, container := range containers {
		stream, err := pods.GetLogs(podName, &corev1.PodLogOptions{Container: container, Follow: true, TailLines: ptr.To[int64](200)}).Stream(ctx)
		if err != nil {
			fmt.Fprintf(w, "data: [%s] %v\n\n", container, err)
			flusher.Flush()
			continue
		}
		scanner := bufio.NewScanner(stream)
		scanner.Buffer(make([]byte, 64*1024), 1<<20)
		for scanner.Scan() {
			fmt.Fprintf(w, "data: %s\n\n", scanner.Text())
			flusher.Flush()
		}
		stream.Close()
	}
	fmt.Fprint(w, "event: end\ndata: \n\n")
	flusher.Flush()
}

func (s *Server) logTarget(ctx context.Context, app apps.App, source string) (string, []string, error) {
	pods := s.Store.Client.CoreV1().Pods(s.Config.Namespace)
	switch source {
	case "build":
		builds, err := s.Builder.Builds(ctx, app.Name)
		if err != nil {
			return "", nil, err
		}
		if len(builds) == 0 {
			return "", nil, errNoLogs
		}
		list, err := pods.List(ctx, metav1.ListOptions{LabelSelector: batchv1.JobNameLabel + "=" + builds[0].Name})
		if err != nil {
			return "", nil, err
		}
		pod, err := newestPod(list.Items)
		return pod, buildContainers, err
	case "runtime":
		list, err := pods.List(ctx, metav1.ListOptions{LabelSelector: runtimeSelector(app.Name)})
		if err != nil {
			return "", nil, err
		}
		pod, err := newestPod(list.Items)
		return pod, []string{"app"}, err
	default:
		return "", nil, fmt.Errorf("%w: source must be build or runtime", errBadRequest)
	}
}

func newestPod(pods []corev1.Pod) (string, error) {
	if len(pods) == 0 {
		return "", errNoLogs
	}
	newest := pods[0]
	for _, pod := range pods[1:] {
		if newest.CreationTimestamp.Before(&pod.CreationTimestamp) {
			newest = pod
		}
	}
	return newest.Name, nil
}
```

```go
	client, err := kubernetes.NewForConfig(restConfig)
	if err != nil {
		return err
	}
	store := &apps.Store{Client: client, Config: config}
	builder := &build.Builder{Client: client, Config: config, Store: store, ResolveHead: build.LsRemote(config.GitHubToken)}
	server := &web.Server{Config: config, Store: store, Builder: builder}
	if config.PGAdminURL != "" {
		server.Provision = pg.Provisioner{AdminURL: config.PGAdminURL}.Provision
	}

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()
	go builder.Run(ctx, 10*time.Second)

	// No WriteTimeout: the log stream stays open while it follows a pod.
	httpServer := &http.Server{Addr: config.Addr, Handler: server.Handler(), ReadHeaderTimeout: 10 * time.Second}
	go func() {
		<-ctx.Done()
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()
		httpServer.Shutdown(shutdownCtx)
	}()
	log.Printf("minirail listening on %s", config.Addr)
	if err := httpServer.ListenAndServe(); !errors.Is(err, http.ErrServerClosed) {
		return err
	}
	return nil
}
```

```sh
wait_for "$app_url/db" "DATABASE_URL set"
```

## Risks

- 🔵 Rootless BuildKit on DOKS kernels is unproven until P5 Step 4; if it fails, the build pod logs say why (usually seccomp/AppArmor or user namespaces).
- 🔵 Cost (~$80/month) is the spec's unverified estimate; P5 Step 3 checks it before any spend.
- 🔵 Registry tier `basic` allows 5 repos: `minirail` + 4 apps. The 5th app's push fails; set `TF_VAR_registry_tier=professional` (+$15/month per spec) and re-run `make up`.
- 🔵 Local Terraform state (`infra/*/terraform.tfstate`) holds DB admin and registry credentials; it is git-ignored, so back it up outside the repo. Losing it means cleaning DO resources by hand.
- 🔵 Builds share nodes with apps; a heavy build can starve small apps. Build limits and autoscaling to 4 nodes mitigate it.
- 🔵 Railpack ships often; CLI and frontend must be bumped together (both pinned to v0.40.1).
- 🔵 Stacked PRs: deleting a base branch before retargeting closes the next PR (see Stacked-PR rules).
